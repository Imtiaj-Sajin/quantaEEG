%% Double-anonymous build of the manuscript, for peer review.
%%
%% This file holds no text of its own. It passes IOP's [anonymous] option to
%% the class and then reads main.tex unchanged, so the named and anonymous
%% versions can never drift apart. The option hides the author list,
%% affiliations, emails, acknowledgements, funding, author contributions and
%% data availability; main.tex's \codestatement and \anonsuppdata switch the
%% body's code link to the anonymised supplementary archive.
%%
%%   latexmk -pdf -outdir=build main_anonymous.tex   -> build/main_anonymous.pdf
%%
%% For upload, paper/make_overleaf_zip.py writes a single main.tex with the
%% option set directly, since IOP's system expects one self-contained file.
\PassOptionsToClass{anonymous}{iopjournal}
\input{main}
